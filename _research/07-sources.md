## Reproduce the GitHub evidence

```bash
python3 - <<'PY'
import json, urllib.request, datetime, time
repos = ["paperclipai/paperclip","crewAIInc/crewAI","langchain-ai/langgraph","FoundationAgents/MetaGPT",
         "OpenBMB/ChatDev","cft0808/edict","Yeachan-Heo/oh-my-claudecode","alookai/alook","5dive-ai/5dive",
         "Miosa-osa/canopy","agems-ai/agems","a2aproject/A2A","modelcontextprotocol/servers"]
out=[]
for r in repos:
    with urllib.request.urlopen(f"https://api.github.com/repos/{r}", timeout=20) as f: d=json.load(f)
    out.append({"repo":d["full_name"],"stars":d["stargazers_count"],"forks":d["forks_count"],
                "license":(d.get("license") or {}).get("spdx_id"),"created":d["created_at"],"pushed":d["pushed_at"]})
    time.sleep(0.6)
print(json.dumps({"captured":datetime.datetime.utcnow().isoformat()+"Z","repos":out}, indent=2))
PY
```
